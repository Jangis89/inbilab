-- ============================================ -- 인비랩 2단계: 유튜브 데이터 테이블 -- Supabase SQL Editor에 붙여넣고 Run 하세요. -- ============================================ -- 1) 채널 표 (수집 대상 채널과 최신 수치) create table if not exists public.channels ( id text primary key, -- 유튜브 채널 ID (UC...) title text not null default '', handle text default '', thumbnail text default '', country text default '', subscriber_count bigint default 0, view_count bigint default 0, video_count integer default 0, daily_views bigint, -- 어제 대비 조회수 증가량 daily_subs bigint, -- 어제 대비 구독자 증가량 is_active boolean not null default true, stats_date date, -- 최신 수치 기준일 added_at timestamptz not null default now() ); -- 2) 채널 일일 기록 표 (그래프와 성장률 계산의 재료) create table if not exists public.channel_snapshots ( channel_id text not null references public.channels (id) on delete cascade, date date not null, subscriber_count bigint default 0, view_count bigint default 0, video_count integer default 0, primary key (channel_id, date) ); -- 3) 인기(로켓) 영상 표 (매일 수집) create table if not exists public.trending_videos ( video_id text not null, date date not null, title text not null default '', channel_id text default '', channel_title text default '', thumbnail text default '', published_at timestamptz, view_count bigint default 0, like_count bigint default 0, is_short boolean default false, rank integer, primary key (video_id, date) ); -- 4) 채널 수집 요청 표 (회원이 "이 채널도 분석해줘" 요청) create table if not exists public.channel_requests ( id bigint generated always as identity primary key, query text not null, user_id uuid references auth.users (id) on delete set null, status text not null default '대기', created_at timestamptz not null default now() ); -- 5) 보안 규칙: 읽기는 누구나 / 쓰기는 수집 로봇(관리자 열쇠)만 alter table public.channels enable row level security; alter table public.channel_snapshots enable row level security; alter table public.trending_videos enable row level security; alter table public.channel_requests enable row level security; drop policy if exists "channels_read" on public.channels; create policy "channels_read" on public.channels for select using (true); drop policy if exists "snapshots_read" on public.channel_snapshots; create policy "snapshots_read" on public.channel_snapshots for select using (true); drop policy if exists "trending_read" on public.trending_videos; create policy "trending_read" on public.trending_videos for select using (true); drop policy if exists "requests_read" on public.channel_requests; drop policy if exists "requests_insert" on public.channel_requests; create policy "requests_read" on public.channel_requests for select using (true); create policy "requests_insert" on public.channel_requests for insert with check (auth.uid() = user_id);
